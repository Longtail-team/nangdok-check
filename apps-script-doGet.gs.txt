/**********************************************************************
 *  전래동화 출석체크 — Apps Script "데이터 API" 버전 doGet
 *
 *  ▸ 이 코드가 기존 doGet(화면을 그려주던 것)을 "대체" 합니다.
 *  ▸ 이제 화면은 Netlify 사이트가 그리고, Apps Script는 데이터(JSON)만 응답 →
 *    구글 회색 배지가 사라집니다.
 *
 *  ── 적용 방법 ──────────────────────────────────────────────────
 *  1) 기존 doGet 함수는 지우기 전에 딴 곳(메모장)에 복사해 백업.
 *  2) doGet 이 들어있는 파일에서 기존 doGet 전체를 아래 코드로 교체.
 *  3) getConfig / getFamilyByPhone / getLeaderboard / submitCheck
 *     함수들은 그대로 둡니다(건드리지 않음).
 *  4) 배포 → 배포 관리 → 편집(연필) → 버전 "새 버전" → 배포.
 *  5) 배포 설정의 액세스 권한은 계속 "모든 사용자(Anyone)" 여야 함.
 *  ※ 과정마다 별도 배포라면, 각 과정 스크립트에 같은 방식으로 적용.
 *********************************************************************/

function doGet(e) {
  var p  = (e && e.parameter) ? e.parameter : {};
  var fn = p.fn;
  var cb = p.callback || 'callback';
  var out;

  try {
    if      (fn === 'getConfig')        out = getConfig();
    else if (fn === 'getFamilyByPhone') out = getFamilyByPhone(p.a0);
    else if (fn === 'getLeaderboard')   out = getLeaderboard();
    else if (fn === 'submitCheck')      out = submitCheck(Number(p.a0), Number(p.a1), p.a2 === 'true');
    else                                out = { ok: false, error: '알 수 없는 요청: ' + fn };
  } catch (err) {
    out = { ok: false, error: String((err && err.message) || err) };
  }

  // JSONP 응답: 브라우저가 <script>로 불러가서 CORS 없이 데이터 수신
  return ContentService
    .createTextOutput(cb + '(' + JSON.stringify(out) + ')')
    .setMimeType(ContentService.MimeType.JAVASCRIPT);
}
